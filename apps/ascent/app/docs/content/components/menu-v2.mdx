## Usage

```tsx
import {
    MenuV2,
    MenuV2Alignment,
    MenuV2Side,
    MenuV2ItemVariant,
    MenuV2ItemActionType,
} from '@juspay/blend-design-system'

function MyComponent() {
    const items = [
        {
            label: 'Actions',
            items: [
                {
                    label: { text: 'Edit', leftSlot: <PencilIcon /> },
                    onClick: () => console.log('Edit'),
                },
                {
                    label: { text: 'Delete', leftSlot: <TrashIcon /> },
                    variant: MenuV2ItemVariant.ACTION,
                    actionType: MenuV2ItemActionType.DANGER,
                    onClick: () => console.log('Delete'),
                },
            ],
        },
    ]

    return (
        <MenuV2
            trigger={<Button text="Open Menu" />}
            items={items}
            alignment={MenuV2Alignment.START}
            side={MenuV2Side.BOTTOM}
        />
    )
}
```
