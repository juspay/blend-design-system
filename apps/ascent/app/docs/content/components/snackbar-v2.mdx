## Usage

```tsx
import {
    SnackbarV2,
    SnackbarV2Position,
    SnackbarV2Variant,
} from '@juspay/blend-design-system'

function App() {
    return (
        <SnackbarV2
            position={SnackbarV2Position.TOP_RIGHT}
            dismissOnClickAway={true}
        />
    )
}

// Show a toast
import { addSnackbarV2 } from '@juspay/blend-design-system'

addSnackbarV2({
    header: 'Success!',
    description: 'Your changes have been saved.',
    variant: SnackbarV2Variant.SUCCESS,
    duration: 5000,
})
```

## Toast Options

```tsx
toast({
    header: 'Warning',
    description: 'Please review your input.',
    variant: SnackbarV2Variant.WARNING,
    duration: 4000,
    position: SnackbarV2Position.BOTTOM_CENTER,
    actionButton: {
        label: 'Undo',
        onClick: () => console.log('Undo clicked'),
        autoDismiss: false,
    },
    onClose: () => console.log('Toast closed'),
})
```

## Component Tokens

You can style the SnackbarV2 component using the following tokens:

```tsx
export type SnackbarV2TokensType = {
    width: CSSObject['width']
    maxWidth: CSSObject['maxWidth']
    minWidth: CSSObject['minWidth']
    backgroundColor: CSSObject['backgroundColor']
    borderRadius: CSSObject['borderRadius']
    padding: CSSObject['padding']
    boxShadow: CSSObject['boxShadow']
    gap: CSSObject['gap']
    slot: {
        height: CSSObject['height']
        width: CSSObject['width']
        color: {
            [key in SnackbarV2Variant]: CSSObject['color']
        }
    }
    mainContainer: {
        gap: CSSObject['gap']
        content: {
            gap: CSSObject['gap']
            textContainer: {
                gap: CSSObject['gap']
                header: {
                    color: {
                        [key in SnackbarV2Variant]: CSSObject['color']
                    }
                    fontSize: CSSObject['fontSize']
                    fontWeight: CSSObject['fontWeight']
                    lineHeight: CSSObject['lineHeight']
                }
                description: {
                    color: {
                        [key in SnackbarV2Variant]: CSSObject['color']
                    }
                    fontSize: CSSObject['fontSize']
                    fontWeight: CSSObject['fontWeight']
                    lineHeight: CSSObject['lineHeight']
                }
            }
            actionContainer: {
                primaryAction: {
                    color: {
                        [key in SnackbarV2Variant]: CSSObject['color']
                    }
                    fontSize: CSSObject['fontSize']
                    fontWeight: CSSObject['fontWeight']
                    lineHeight: CSSObject['lineHeight']
                }
            }
        }
        closeButton: {
            height: CSSObject['height']
            color: {
                [key in SnackbarV2Variant]: CSSObject['color']
            }
        }
    }
}
```
