## Usage

```tsx
import { SearchInputV2 } from '@juspay/blend-design-system'

function MyComponent() {
    const [query, setQuery] = useState('')

    return (
        <SearchInputV2
            placeholder="Search products..."
            value={query}
            onChange={(e) => setQuery(e.target.value)}
            allowClear={true}
        />
    )
}
```
