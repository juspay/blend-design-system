## Usage

```tsx
import {
    CodeEditorV2,
    CodeEditorV2Variant,
    MonacoTheme,
} from '@juspay/blend-design-system'

function MyComponent() {
    const [code, setCode] = useState('const x = 1;')

    return (
        <CodeEditorV2
            value={code}
            onChange={setCode}
            language="typescript"
            showLineNumbers={true}
            header={{
                showHeader: true,
                title: 'main.ts',
                showCopyButton: true,
            }}
        />
    )
}
```

## Diff Mode Usage

```tsx
import { CodeEditorV2, CodeEditorV2Variant } from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <CodeEditorV2
            variant={CodeEditorV2Variant.DIFF}
            value="const newValue = 2;"
            originalValue="const oldValue = 1;"
            language="typescript"
            renderSideBySide={true}
            isDiffUnchangedCollapsed={true}
        />
    )
}
```

## Component Tokens

You can style the CodeEditorV2 component using the following tokens:

```tsx
export type CodeEditorV2Tokens = {
    container: {
        borderRadius: CSSObject['borderRadius']
        border: CSSObject['border']
        background: CSSObject['background']
    }
    header: {
        padding: CSSObject['padding']
        background: CSSObject['background']
        borderBottom: CSSObject['borderBottom']
        title: {
            fontSize: CSSObject['fontSize']
            fontWeight: CSSObject['fontWeight']
            color: CSSObject['color']
        }
    }
    editor: {
        fontFamily: CSSObject['fontFamily']
        fontSize: CSSObject['fontSize']
        lineHeight: CSSObject['lineHeight']
    }
    copyButton: {
        color: CSSObject['color']
        hoverColor: CSSObject['color']
    }
}
```
