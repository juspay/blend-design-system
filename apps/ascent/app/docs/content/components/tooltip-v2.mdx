## Usage

```tsx
import {
    TooltipV2,
    TooltipV2Side,
    TooltipV2Align,
    TooltipV2Size,
    TooltipV2SlotDirection,
} from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <TooltipV2
            content="This is helpful information"
            side={TooltipV2Side.TOP}
            align={TooltipV2Align.CENTER}
            size={TooltipV2Size.MD}
            showArrow={true}
        >
            <Button text="Hover me" />
        </TooltipV2>
    )
}
```

## With Slot

```tsx
import { TooltipV2, TooltipV2SlotDirection } from '@juspay/blend-design-system'
;<TooltipV2
    content="More details available"
    slot={<InfoIcon />}
    slotDirection={TooltipV2SlotDirection.RIGHT}
>
    <span>Status</span>
</TooltipV2>
```

    background: CSSObject['backgroundColor']
