## Usage

```tsx
import { OTPInputV2 } from '@juspay/blend-design-system'
import { useState } from 'react'

function MyComponent() {
    const [otp, setOtp] = useState('')

    return <OTPInputV2 length={6} value={otp} onChange={setOtp} />
}
```

```tsx

    container: {
        gap: CSSObject['gap']
    }
    input: {
        width: {
            [key in InputSizeV2]: CSSObject['width']
        }
        height: {
            [key in InputSizeV2]: CSSObject['height']
        }
        borderRadius: CSSObject['borderRadius']
        border: {
            default: CSSObject['border']
            focus: CSSObject['border']
            error: CSSObject['border']
            disabled: CSSObject['border']
        }
        background: {
            default: CSSObject['background']
            disabled: CSSObject['background']
        }
        fontSize: CSSObject['fontSize']
        fontWeight: CSSObject['fontWeight']
        textAlign: CSSObject['textAlign']
        color: CSSObject['color']
    }
}
```
