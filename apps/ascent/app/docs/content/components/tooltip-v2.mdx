## Usage

```tsx
import {
    TooltipV2,
    TooltipV2Side,
    TooltipV2Align,
    TooltipV2Size,
    TooltipV2SlotDirection,
} from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <TooltipV2
            content="This is helpful information"
            side={TooltipV2Side.TOP}
            align={TooltipV2Align.CENTER}
            size={TooltipV2Size.MD}
            showArrow={true}
        >
            <Button text="Hover me" />
        </TooltipV2>
    )
}
```

## With Slot

```tsx
import { TooltipV2, TooltipV2SlotDirection } from '@juspay/blend-design-system'
;<TooltipV2
    content="More details available"
    slot={<InfoIcon />}
    slotDirection={TooltipV2SlotDirection.RIGHT}
>
    <span>Status</span>
</TooltipV2>
```

## Component Tokens

You can style the TooltipV2 component using the following tokens:

```tsx
export type TooltipV2TokensType = {
    content: {
        padding: {
            [key in TooltipV2Size]: CSSObject['padding']
        }
        borderRadius: CSSObject['borderRadius']
        background: CSSObject['background']
        boxShadow: CSSObject['boxShadow']
        maxWidth: {
            [key in TooltipV2Size]: CSSObject['maxWidth']
        }
    }
    text: {
        fontSize: CSSObject['fontSize']
        fontWeight: CSSObject['fontWeight']
        color: CSSObject['color']
        lineHeight: CSSObject['lineHeight']
    }
    arrow: {
        size: CSSObject['width']
        height: CSSObject['height']
        background: CSSObject['background']
    }
    slot: {
        gap: CSSObject['gap']
    }
}
```
