## Usage

```tsx
import {
    StepperV2,
    StepperV2StepStatus,
    StepperV2Type,
} from '@juspay/blend-design-system'

function MyComponent() {
    const steps = [
        {
            id: 1,
            title: 'Personal Info',
            status: StepperV2StepStatus.COMPLETED,
        },
        {
            id: 2,
            title: 'Account Details',
            status: StepperV2StepStatus.CURRENT,
        },
        {
            id: 3,
            title: 'Review',
            status: StepperV2StepStatus.PENDING,
        },
    ]

    return (
        <StepperV2
            steps={steps}
            stepperType={StepperV2Type.HORIZONTAL}
            clickable={true}
            onStepClick={(index) => console.log('Clicked step:', index)}
        />
    )
}
```

## With Substeps

```tsx
import {
    StepperV2,
    StepperV2StepStatus,
} from '@juspay/blend-design-system'

const steps = [
    {
        id: 1,
        title: 'Setup',
        status: StepperV2StepStatus.COMPLETED,
        substeps: [
            { id: 11, title: 'Configure', status: StepperV2StepStatus.COMPLETED },
            { id: 12, title: 'Verify', status: StepperV2StepStatus.COMPLETED },
        ],
        isExpandable: true,
    },
    {
        id: 2,
        title: 'Deploy',
        status: StepperV2StepStatus.CURRENT,
    },
]

<StepperV2
    steps={steps}
    onSubstepClick={(stepId, substepIndex) => console.log(stepId, substepIndex)}
/>
```
