## Usage

```tsx
import {
    SnackbarV2,
    SnackbarV2Position,
    SnackbarV2Variant,
} from '@juspay/blend-design-system'

function App() {
    return (
        <SnackbarV2
            position={SnackbarV2Position.TOP_RIGHT}
            dismissOnClickAway={true}
        />
    )
}

// Show a toast
import { addSnackbarV2 } from '@juspay/blend-design-system'

addSnackbarV2({
    header: 'Success!',
    description: 'Your changes have been saved.',
    variant: SnackbarV2Variant.SUCCESS,
    duration: 5000,
})
```

## Toast Options

```tsx
toast({
    header: 'Warning',
    description: 'Please review your input.',
    variant: SnackbarV2Variant.WARNING,
    duration: 4000,
    position: SnackbarV2Position.BOTTOM_CENTER,
    actionButton: {
        label: 'Undo',
        onClick: () => console.log('Undo clicked'),
        autoDismiss: false,
    },
    onClose: () => console.log('Toast closed'),
})
```

## Persistent Toasts

Set `duration: Infinity` for a toast that never auto-dismisses.

```tsx
import { addSnackbarV2, SnackbarV2Variant } from '@juspay/blend-design-system'

addSnackbarV2({
    header: 'Upload failed',
    description: 'invoice-2026-Q1.pdf could not be uploaded.',
    variant: SnackbarV2Variant.ERROR,
    duration: Infinity,
    actionButton: {
        label: 'Retry',
        autoDismiss: false, // clicking Retry keeps the toast open
        onClick: retryUpload,
    },
})
```

`actionButton.autoDismiss` is separate from the toast duration. Set it to `false` when
clicking the action should keep the persistent toast open. The close button remains
available.

A persistent toast is exempt from `dismissOnClickAway`, so clicking elsewhere on the page
closes ordinary toasts but leaves it open. It also does not count against the visible-toast
limit, so newer toasts cannot push it out of view.

It is still a toast: it overlays the page, supports a single action, and is announced once
by screen readers. For a message the user must not miss, use `AlertV2` (inline, in reading
order, supports a primary and secondary action) or `ModalV2` (blocking).

### Do not use `duration: 0`

`duration: 0` does **not** create a persistent toast. `0` is falsy, so it falls through to
the 4000ms default and you get an ordinary 4-second toast.

### Choosing the right component

| Message                                                                               | Component                              |
| ------------------------------------------------------------------------------------- | -------------------------------------- |
| Transient and passive — "Saved", "Copied"                                             | `SnackbarV2`                           |
| Optional action, no deadline — "Upload failed / Retry", "Undo"                        | `SnackbarV2` with `duration: Infinity` |
| The user must act, or ignoring it has a consequence — session expiry, unsaved changes | `AlertV2` (inline)                     |
| The user must act before continuing                                                   | `ModalV2` (blocking)                   |
| Must survive navigation or be re-findable later                                       | Neither — needs a notification centre  |

Snackbars support a single action and live in a stack, which makes them a poor host for
anything with a deadline or a consequence. Reach for `AlertV2` there: it is inline, always
visible, and supports a primary and a secondary action.

## Component Tokens

You can style the SnackbarV2 component using the following tokens:

```tsx
export type SnackbarV2TokensType = {
    width: CSSObject['width']
    maxWidth: CSSObject['maxWidth']
    minWidth: CSSObject['minWidth']
    backgroundColor: CSSObject['backgroundColor']
    borderRadius: CSSObject['borderRadius']
    padding: CSSObject['padding']
    boxShadow: CSSObject['boxShadow']
    gap: CSSObject['gap']
    slot: {
        height: CSSObject['height']
        width: CSSObject['width']
        color: {
            [key in SnackbarV2Variant]: CSSObject['color']
        }
    }
    mainContainer: {
        gap: CSSObject['gap']
        content: {
            gap: CSSObject['gap']
            textContainer: {
                gap: CSSObject['gap']
                header: {
                    color: {
                        [key in SnackbarV2Variant]: CSSObject['color']
                    }
                    fontSize: CSSObject['fontSize']
                    fontWeight: CSSObject['fontWeight']
                    lineHeight: CSSObject['lineHeight']
                }
                description: {
                    color: {
                        [key in SnackbarV2Variant]: CSSObject['color']
                    }
                    fontSize: CSSObject['fontSize']
                    fontWeight: CSSObject['fontWeight']
                    lineHeight: CSSObject['lineHeight']
                }
            }
            actionContainer: {
                primaryAction: {
                    color: {
                        [key in SnackbarV2Variant]: CSSObject['color']
                    }
                    fontSize: CSSObject['fontSize']
                    fontWeight: CSSObject['fontWeight']
                    lineHeight: CSSObject['lineHeight']
                }
            }
        }
        closeButton: {
            height: CSSObject['height']
            color: {
                [key in SnackbarV2Variant]: CSSObject['color']
            }
        }
    }
}
```
