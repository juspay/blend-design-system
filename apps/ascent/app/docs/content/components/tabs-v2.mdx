## Usage

```tsx
import {
    TabsV2,
    TabsV2List,
    TabsV2Trigger,
    TabsV2Content,
    TabsV2Variant,
    TabsV2Size,
} from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <TabsV2
            defaultValue="tab1"
            variant={TabsV2Variant.BOXED}
            size={TabsV2Size.MD}
        >
            <TabsV2List>
                <TabsV2Trigger value="tab1">Overview</TabsV2Trigger>
                <TabsV2Trigger value="tab2">Details</TabsV2Trigger>
                <TabsV2Trigger value="tab3">Settings</TabsV2Trigger>
            </TabsV2List>
            <TabsV2Content value="tab1">Overview content</TabsV2Content>
            <TabsV2Content value="tab2">Details content</TabsV2Content>
            <TabsV2Content value="tab3">Settings content</TabsV2Content>
        </TabsV2>
    )
}
```

## With Tab Items

```tsx
import { TabsV2, TabsV2Variant } from '@juspay/blend-design-system'

const tabItems = [
    {
        value: 'dashboard',
        label: 'Dashboard',
        content: <DashboardView />,
        leftSlot: <ChartIcon />,
    },
    {
        value: 'users',
        label: 'Users',
        content: <UsersView />,
        rightSlot: <UserCountBadge />,
    },
]

<TabsV2 variant={TabsV2Variant.FLOATING} expanded={true}>
    {tabItems.map((item) => (
        <TabsV2.Tab key={item.value} {...item} />
    ))}
</TabsV2>
```

## Component Tokens

You can style the TabsV2 component using the following tokens:

```tsx
export type TabsV2TokensType = {
    list: {
        gap: CSSObject['gap']
        padding: CSSObject['padding']
        borderRadius: CSSObject['borderRadius']
        background: CSSObject['background']
    }
    trigger: {
        padding: CSSObject['padding']
        borderRadius: CSSObject['borderRadius']
        fontSize: CSSObject['fontSize']
        fontWeight: CSSObject['fontWeight']
        gap: CSSObject['gap']
        colors: {
            default: CSSObject['color']
            hover: CSSObject['color']
            active: CSSObject['color']
            disabled: CSSObject['color']
        }
        backgrounds: {
            default: CSSObject['background']
            hover: CSSObject['background']
            active: CSSObject['background']
        }
        borders: {
            default: CSSObject['border']
            active: CSSObject['border']
        }
    }
    variants: {
        boxed: {
            listBackground: CSSObject['background']
            triggerActiveBackground: CSSObject['background']
        }
        floating: {
            listBackground: CSSObject['background']
            shadow: CSSObject['boxShadow']
        }
        underline: {
            borderBottom: CSSObject['borderBottom']
            activeBorder: CSSObject['borderBottom']
        }
        pills: {
            borderRadius: CSSObject['borderRadius']
        }
    }
    content: {
        padding: CSSObject['padding']
    }
}
```
