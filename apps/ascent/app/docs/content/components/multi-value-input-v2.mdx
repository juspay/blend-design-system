## Usage

```tsx
import { MultiValueInputV2, InputSizeV2 } from '@juspay/blend-design-system'

function MyComponent() {
    const [values, setValues] = useState(['tag1', 'tag2'])

    return (
        <MultiValueInputV2
            label="Tags"
            placeholder="Add tags..."
            values={values}
            onChange={setValues}
            size={InputSizeV2.MD}
            separator=","
        />
    )
}
```
