## Usage

```tsx
import { DrawerV2 } from '@juspay/blend-design-system'

function MyComponent() {
    const [open, setOpen] = useState(false)

    return (
        <DrawerV2
            open={open}
            onOpenChange={setOpen}
            direction="right"
            size="md"
        >
            <div>
                <h2>Drawer Content</h2>
                <p>Your content here</p>
            </div>
        </DrawerV2>
    )
}
```

## Component Tokens

You can style the DrawerV2 component using the following tokens:

```tsx
export type DrawerV2TokensType = {
    overlay: {
        backgroundColor: CSSObject['backgroundColor']
        backdropFilter: CSSObject['backdropFilter']
    }
    content: {
        backgroundColor: CSSObject['backgroundColor']
        boxShadow: CSSObject['boxShadow']
    }
    sizes: {
        sm: CSSObject['width']
        md: CSSObject['width']
        lg: CSSObject['width']

}
```
