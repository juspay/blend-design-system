## Usage

```tsx
import { NumberInputV2, InputSizeV2 } from '@juspay/blend-design-system'

function MyComponent() {
    const [value, setValue] = useState(0)

    return (
        <NumberInputV2
            label="Quantity"
            value={value}
            onChange={setValue}
            min={0}
            max={100}
            step={1}
            size={InputSizeV2.MD}
        />
    )
}
```
