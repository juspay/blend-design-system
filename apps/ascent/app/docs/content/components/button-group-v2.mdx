## Usage

```tsx
import {
    ButtonGroupV2,
    ButtonV2,
    ButtonV2Type,
    ButtonV2Size,
} from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <ButtonGroupV2 stacked={false}>
            <ButtonV2
                text="Cancel"
                buttonType={ButtonV2Type.SECONDARY}
                size={ButtonV2Size.MEDIUM}
                onClick={() => console.log('Cancel clicked')}
            />
            <ButtonV2
                text="Save Changes"
                buttonType={ButtonV2Type.PRIMARY}
                size={ButtonV2Size.MEDIUM}
                onClick={() => console.log('Save clicked')}
            />
        </ButtonGroupV2>
    )
}
```

## Stacked Layout

```tsx
import {
    ButtonGroupV2,
    ButtonV2,
    ButtonV2Type,
} from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <ButtonGroupV2 stacked={true}>
            <ButtonV2 text="Option 1" buttonType={ButtonV2Type.PRIMARY} />
            <ButtonV2 text="Option 2" buttonType={ButtonV2Type.SECONDARY} />
            <ButtonV2 text="Option 3" buttonType={ButtonV2Type.SECONDARY} />
        </ButtonGroupV2>
    )
}
```

## With Custom Gap

```tsx
import { ButtonGroupV2, ButtonV2 } from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <ButtonGroupV2 gap={16}>
            <ButtonV2 text="Cancel" />
            <ButtonV2 text="Submit" />
        </ButtonGroupV2>
    )
}
```

    gap: {
        horizontal: CSSObject['gap']
