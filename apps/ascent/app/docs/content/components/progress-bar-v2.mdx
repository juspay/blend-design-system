## Usage

```tsx
import {
    ProgressBarV2,
    ProgressBarV2Size,
    ProgressBarV2Variant,
    ProgressBarV2Appearance,
} from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <ProgressBarV2
            value={75}
            min={0}
            max={100}
            size={ProgressBarV2Size.MD}
            variant={ProgressBarV2Variant.LINEAR}
            appearance={ProgressBarV2Appearance.SOLID}
            showLabel={true}
        />
    )
}
```

## Circular Progress

```tsx
import {
    ProgressBarV2,
    ProgressBarV2Variant,
} from '@juspay/blend-design-system'
;<ProgressBarV2
    value={60}
    variant={ProgressBarV2Variant.CIRCULAR}
    size={ProgressBarV2Size.LG}
    showLabel={true}
/>
```

## Component Tokens

You can style the ProgressBarV2 component using the following tokens:

```tsx
export type ProgressBarV2TokenType = {
    linear: {
        track: {
            height: {
                [key in ProgressBarV2Size]: CSSObject['height']
            }
            background: CSSObject['background']
            borderRadius: CSSObject['borderRadius']
        }
        fill: {
            background: CSSObject['background']
            borderRadius: CSSObject['borderRadius']
            transition: CSSObject['transition']
        }
        segmented: {
            gap: CSSObject['gap']
            segmentWidth: CSSObject['width']
            segmentBorderRadius: CSSObject['borderRadius']
        }
    }
    circular: {
        sizes: {
            [key in ProgressBarV2Size]: CSSObject['width']
        }
        track: {
            stroke: CSSObject['stroke']
            strokeWidth: CSSObject['strokeWidth']
        }
        fill: {
            stroke: CSSObject['stroke']
            strokeWidth: CSSObject['strokeWidth']
            strokeLinecap: CSSObject['strokeLinecap']
            transition: CSSObject['transition']
        }
    }
    label: {
        fontSize: CSSObject['fontSize']
        fontWeight: CSSObject['fontWeight']
        color: CSSObject['color']
        margin: CSSObject['margin']
    }
}
```
