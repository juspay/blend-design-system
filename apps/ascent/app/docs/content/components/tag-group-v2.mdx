## Usage

```tsx
import {
    TagGroupV2,
    TagV2,
    TagV2Color,
    TagV2Size,
} from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <TagGroupV2 stacked={false} gap="8px">
            <TagV2
                text="React"
                size={TagV2Size.SM}
                color={TagV2Color.PRIMARY}
            />
            <TagV2
                text="TypeScript"
                size={TagV2Size.SM}
                color={TagV2Color.SUCCESS}
            />
            <TagV2
                text="NextJS"
                size={TagV2Size.SM}
                color={TagV2Color.NEUTRAL}
            />
            <TagV2
                text="Tailwind"
                size={TagV2Size.SM}
                color={TagV2Color.WARNING}
            />
        </TagGroupV2>
    )
}
```

## Stacked Layout

```tsx
import { TagGroupV2, TagV2 } from '@juspay/blend-design-system'
;<TagGroupV2 stacked={true} gap={4}>
    <TagV2 text="Feature 1" />
    <TagV2 text="Feature 2" />
    <TagV2 text="Feature 3" />
</TagGroupV2>
```
