## Usage

```tsx
import { OTPInputV2, InputSizeV2 } from '@juspay/blend-design-system'

function MyComponent() {
    const [otp, setOtp] = useState('')

    return (
        <OTPInputV2
            length={6}
            value={otp}
            onChange={setOtp}
            size={InputSizeV2.MD}
            onComplete={(value) => console.log('OTP entered:', value)}
        />
    )
}
```
