## Usage

```tsx
import { SidebarV2 } from '@juspay/blend-design-system'

function MyComponent() {
    const directoryData = [
        {
            id: 'home',
            label: 'Home',
            icon: <HomeIcon />,
            href: '/',
        },
        {
            id: 'settings',
            label: 'Settings',
            icon: <SettingsIcon />,
            children: [
                { id: 'profile', label: 'Profile', href: '/settings/profile' },
                {
                    id: 'security',
                    label: 'Security',
                    href: '/settings/security',
                },
            ],
        },
    ]

    const secondaryItems = [
        { label: 'Overview', value: 'overview', icon: <ChartIcon /> },
        { label: 'Analytics', value: 'analytics', icon: <GraphIcon /> },
    ]

    return (
        <SidebarV2
            data={directoryData}
            secondarySidebar={{
                items: secondaryItems,
                selected: 'overview',
                onSelect: (value) => console.log(value),
            }}
            isExpanded={true}
            onExpandedChange={(expanded) => console.log(expanded)}
        />
    )
}
```

## Component Tokens

You can style the SidebarV2 component using the following tokens:

```tsx
export type SidebarV2TokensType = {
    container: {
        background: CSSObject['background']
        borderRight: CSSObject['borderRight']
        width: {
            collapsed: CSSObject['width']
            expanded: CSSObject['width']
        }
    }
    secondaryRail: {
        background: CSSObject['background']
        borderRight: CSSObject['borderRight']
        width: CSSObject['width']
    }
    item: {
        padding: CSSObject['padding']
        borderRadius: CSSObject['borderRadius']
        gap: CSSObject['gap']
        fontSize: CSSObject['fontSize']
        color: {
            default: CSSObject['color']
            hover: CSSObject['color']
            active: CSSObject['color']
        }
        background: {
            default: CSSObject['background']
            hover: CSSObject['background']
            active: CSSObject['background']
        }
    }
    collapseButton: {
        size: CSSObject['width']
        background: CSSObject['background']
        border: CSSObject['border']
        color: CSSObject['color']
    }
    topbar: {
        height: CSSObject['height']
        background: CSSObject['background']
        borderBottom: CSSObject['borderBottom']
        padding: CSSObject['padding']
    }
}
```
