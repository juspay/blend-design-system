## Usage

```tsx
import { TextInputV2, InputSizeV2 } from '@juspay/blend-design-system'

function MyComponent() {
    const [value, setValue] = useState('')

    return (
        <TextInputV2
            label="Email Address"
            placeholder="Enter your email"
            value={value}
            onChange={(e) => setValue(e.target.value)}
            size={InputSizeV2.MD}
            hintText="We'll never share your email"
        />
    )
}
```

## With Dropdown

```tsx
import { TextInputV2, DropdownPosition } from '@juspay/blend-design-system'
;<TextInputV2
    label="Country"
    placeholder="Select or type"
    value={value}
    onChange={handleChange}
    dropdown={{
        position: DropdownPosition.LEFT,
        items: [
            {
                groupLabel: 'Regions',
                items: [
                    { label: 'North America', value: 'na' },
                    { label: 'Europe', value: 'eu' },
                ],
            },
        ],
        selected: selectedRegion,
        onSelect: setSelectedRegion,
    }}
/>
```

## With Error State

```tsx
import { TextInputV2 } from '@juspay/blend-design-system'
;<TextInputV2
    label="Username"
    placeholder="Enter username"
    value={username}
    onChange={handleChange}
    error={{
        show: username.length < 3,
        message: 'Username must be at least 3 characters',
    }}
/>
```

## Component Tokens

```tsx

    container: {
        gap: CSSObject['gap']
    }
    label: {
        fontSize: CSSObject['fontSize']
        fontWeight: CSSObject['fontWeight']
        color: CSSObject['color']
    }
    subLabel: {
        fontSize: CSSObject['fontSize']
        color: CSSObject['color']
    }
    input: {
        height: {
            [key in InputSizeV2]: CSSObject['height']
        }
        padding: CSSObject['padding']
        borderRadius: CSSObject['borderRadius']
        border: {
            default: CSSObject['border']
            hover: CSSObject['border']
            focus: CSSObject['border']
            error: CSSObject['border']
            disabled: CSSObject['border']
        }
        background: {
            default: CSSObject['background']
            disabled: CSSObject['background']
        }
        fontSize: CSSObject['fontSize']
        color: CSSObject['color']
    }
    hintText: {
        fontSize: CSSObject['fontSize']
        color: CSSObject['color']
    }
    errorText: {
        fontSize: CSSObject['fontSize']
        color: CSSObject['color']
    }
    slot: {
        gap: CSSObject['gap']
    }
}
```
