## Usage

```tsx
import { Badge, BadgeColor, BadgeSize } from '@juspay/blend-design-system'

function MyComponent() {
    return (
        // Standalone badge with count
        <Badge count={5} color={BadgeColor.ALERT} size={BadgeSize.MD} />

        // Badge with text
        <Badge text="New" color={BadgeColor.SUCCESS} />

        // Badge as dot (no content)
        <Badge color={BadgeColor.ALERT} />

        // Badge wrapping an element
        <Badge count={10} position="top-right">
            <NotificationIcon />
        </Badge>
    )
}
```

## Component Tokens

You can style the Badge component using the following tokens:

```tsx

    dot: {
        width: {
            [key in BadgeSize]: CSSObject['width']

            [key in BadgeSize]: CSSObject['height']

        borderRadius: CSSObject['borderRadius']
    }
    pill: {
        minWidth: {

        }

            [key in BadgeSize]: CSSObject['paddingLeft']
        }
        paddingRight: {
            [key in BadgeSize]: CSSObject['paddingRight']
        }
        borderRadius: {
            [key in BadgeSize]: CSSObject['borderRadius']

            [key in BadgeSize]: CSSObject['fontSize']

}
```
