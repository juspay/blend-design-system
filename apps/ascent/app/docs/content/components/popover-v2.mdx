## Usage

```tsx
import {
    PopoverV2,
    PopoverV2Size,
    PopoverV2Side,
    PopoverV2Align,
    ButtonType,
} from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <PopoverV2
            heading="Popover Title"
            description="This is a helpful description."
            trigger={
                <Button text="Open Popover" buttonType={ButtonType.PRIMARY} />
            }
            size={PopoverV2Size.MD}
            side={PopoverV2Side.BOTTOM}
            align={PopoverV2Align.CENTER}
            primaryAction={{
                text: 'Confirm',
                buttonType: ButtonType.PRIMARY,
                onClick: () => console.log('Confirmed'),
            }}
            secondaryAction={{
                text: 'Cancel',
                buttonType: ButtonType.SECONDARY,
                onClick: () => console.log('Cancelled'),
            }}
        >
            <div>Additional content goes here</div>
        </PopoverV2>
    )
}
```

## Component Tokens

You can style the PopoverV2 component using the following tokens:

```tsx
export type PopoverV2TokensType = {
    content: {
        background: CSSObject['background']
        borderRadius: CSSObject['borderRadius']
        border: CSSObject['border']
        boxShadow: CSSObject['boxShadow']
        padding: CSSObject['padding']
    }
    header: {
        padding: CSSObject['padding']
        borderBottom: CSSObject['borderBottom']
        gap: CSSObject['gap']
    }
    heading: {
        fontSize: CSSObject['fontSize']
        fontWeight: CSSObject['fontWeight']
        color: CSSObject['color']
    }
    description: {
        fontSize: CSSObject['fontSize']
        color: CSSObject['color']
    }
    body: {
        padding: CSSObject['padding']
    }
    footer: {
        padding: CSSObject['padding']
        borderTop: CSSObject['borderTop']
        gap: CSSObject['gap']
    }
    closeButton: {
        color: CSSObject['color']
        size: CSSObject['width']
    }
    sizes: {
        sm: CSSObject['width']
        md: CSSObject['width']
        lg: CSSObject['width']
    }
}
```
