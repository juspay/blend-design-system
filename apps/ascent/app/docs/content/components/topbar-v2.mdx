## Usage

```tsx
import { TopbarV2, ButtonV2 } from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <TopbarV2
            showBackButton
            onBackClick={() => history.back()}
            rightActions={<ButtonV2 text="Create" />}
        >
            Dashboard
        </TopbarV2>
    )
}
```
