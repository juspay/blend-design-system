## Usage

```tsx
import { ChartV2 } from '@juspay/blend-design-system'
import Highcharts from 'highcharts'

function MyComponent() {
    const options = {
        title: {
            text: 'Monthly Sales',
        },
        series: [
            {
                type: 'line',
                name: 'Sales',
                data: [1, 2, 3, 4, 5],
            },
        ],
    }

    return (
        <ChartV2
            highcharts={Highcharts}
            options={options}
            skeleton={{ show: false }}
            noData={{
                title: 'No Data Available',
                subtitle: 'Try adjusting your filters',
            }}
        />
    )
}
```

## ChartContainerV2

A wrapper component for chart layouts:

```tsx
import { ChartContainerV2, ChartHeaderV2 } from '@juspay/blend-design-system'
;<ChartContainerV2>
    <ChartHeaderV2>
        <h3>Sales Overview</h3>
    </ChartHeaderV2>
    <ChartV2 options={options} />
</ChartContainerV2>
```

## ChartV2Legend

Custom legend component for chart interactivity:

```tsx
import { ChartV2Legend } from '@juspay/blend-design-system'
;<ChartV2Legend
    chartRef={chartRef}
    customLegendItems={[
        { key: 'sales', name: 'Sales', color: '#00A6FF', value: 1200 },
        { key: 'visits', name: 'Visits', color: '#10B981', value: 3400 },
    ]}
    layout="horizontal"
/>
```

## ChartV2Fullscreen

Wrapper component for fullscreen chart functionality:

```tsx
import { ChartV2Fullscreen, ChartV2 } from '@juspay/blend-design-system'
;<ChartV2Fullscreen>
    <ChartV2 options={options} />
</ChartV2Fullscreen>
```

## ChartV2Skeleton

Standalone skeleton loader for chart loading states:

```tsx
import { ChartV2Skeleton } from '@juspay/blend-design-system'
;<ChartV2Skeleton show={true} variant="rectangular" height={300} />
```
