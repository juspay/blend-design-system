## Usage

```tsx
import { Topbar, Button } from '@juspay/blend-design-system'

function MyComponent() {
    return (
        <Topbar
            showBackButton
            onBackClick={() => history.back()}
            rightActions={<Button text="Create" />}
        >
            Dashboard
        </Topbar>
    )
}
```
